ESP32 Firmware Code (HX711 + MQ-5 Sensor)

Flash this sketch onto your ESP32 board. It reads the HX711 40kg load cell and MQ sensor, calculates net fuel, and streams JSON data to this dashboard over WebSockets or HTTP. 

#include <WiFi.h>
#include <Firebase_ESP_Client.h> // Install Firebase Arduino Client Library
#include <HX711.h>

// Wi-Fi Configuration
const char* ssid     = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// Firebase Cloud Configuration
#define API_KEY "YOUR_FIREBASE_API_KEY"
#define DATABASE_URL "https://YOUR_PROJECT_ID-default-rtdb.firebaseio.com"

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// Pinout Definitions
const int LOADCELL_DOUT_PIN = 16;
const int LOADCELL_SCK_PIN  = 4;
const int MQ5_ANALOG_PIN    = 34;

HX711 scale;

// Calibration Parameters
const float TARE_SHELL_KG = 15.30;
const float CALIBRATION_FACTOR = -22800.0;

unsigned long lastTransmit = 0;

void setup() {
  Serial.begin(115200);
  scale.begin(LOADCELL_DOUT_PIN, LOADCELL_SCK_PIN);
  scale.set_scale(CALIBRATION_FACTOR);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");

  // Initialize Firebase Cloud Access
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;
  config.signer.test_mode = true; // Use anonymous access for testing
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  // Send telemetry every 2000ms to Firebase Cloud
  if (millis() - lastTransmit > 2000) {
    lastTransmit = millis();

    float grossWeight = scale.get_units(5);
    if (grossWeight < 0) grossWeight = 0;
    
    int rawMq = analogRead(MQ5_ANALOG_PIN);
    int ppm = map(rawMq, 200, 3500, 40, 800);

    // Upload live data to Firebase Realtime Database
    Firebase.RTDB.setFloat(&fbdo, "/lpg_telemetry/gross_weight", grossWeight);
    Firebase.RTDB.setInt(&fbdo, "/lpg_telemetry/ppm", ppm);
    Firebase.RTDB.setInt(&fbdo, "/lpg_telemetry/rssi", WiFi.RSSI());

    Serial.println("Telemetry dispatched to Firebase Cloud!");
  }
}
